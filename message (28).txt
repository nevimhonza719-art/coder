-- Services
local Lighting = game:GetService("Lighting")

-- Fetch Linoria UI Library
local repo = 'https://raw.githubusercontent.com/violin-suzutsuki/LinoriaLib/main/'
local Library = loadstring(game:HttpGet(repo .. 'Library.lua'))()
local ThemeManager = loadstring(game:HttpGet(repo .. 'addons/ThemeManager.lua'))()
local SaveManager = loadstring(game:HttpGet(repo .. 'addons/SaveManager.lua'))()

-- Ensure all post-processing effects exist locally
local function getOrInitEffect(className)
    local effect = Lighting:FindFirstChildOfClass(className)
    if not effect then
        effect = Instance.new(className)
        effect.Parent = Lighting
    end
    return effect
end

local atmosphere = getOrInitEffect("Atmosphere")
local bloom = getOrInitEffect("BloomEffect")
local blur = getOrInitEffect("BlurEffect")
local colorCorrection = getOrInitEffect("ColorCorrectionEffect")
local sunRays = getOrInitEffect("SunRaysEffect")

-- Create Window
local Window = Library:CreateWindow({
    Title = 'OP Lighting Studio | Unnamed Style',
    Center = true,
    AutoShow = true,
    TabPadding = 8,
    MenuFadeTime = 0.2
})

-- Tabs
local Tabs = {
    Global = Window:AddTab('Global Lighting'),
    PostFX = Window:AddTab('Post-Processing'),
    ['UI Settings'] = Window:AddTab('UI Settings'),
}

-------------------------------------------------------------------------------

-------------------------------------------------------------------------------

-------------------------------------------------------------------------------
-- CONFIGS & SYSTEM LOADER
-------------------------------------------------------------------------------
ThemeManager:SetLibrary(Library)
SaveManager:SetLibrary(Library)
SaveManager:IgnoreThemeSettings()
SaveManager:SetIgnoreKeys({ 'MenuKeybind' })
ThemeManager:SetFolder('LightingStudioSettings')
SaveManager:SetFolder('LightingStudioSettings/Configs')

SaveManager:BuildConfigSection(Tabs['UI Settings'])

SaveManager:LoadAutoloadConfig()

Library:Notify("Ultimate Lighting Studio Initialized! Press Right Shift to toggle.", 6)
